pipeline{
  agent any
  tools{
     maven 'MAVEN-HOME'
  }
  stages{
     stage('git repo & clean'){
        steps{
           //bat "rmdir /s /q mavenjava"
           bat "git clone provide your github link"
           bat "mvn clean -f Maven"
        }
    }
    stage('install'){
       steps {
          bat "mvn install -f Maven"
         }
      }
      stage('test'){
          steps {
            bat "mvn test -f Maven"
          }
       }
       stage('package'){
          steps {
            bat "mvn package -f Maven"
          }
       }
    }
}
